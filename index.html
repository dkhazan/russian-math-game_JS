<script>

const letters = "АБВГДЕЁЖЗИЙКЛМНОПРСТУФХЦЧШЩЪЫЬЭЮЯ".split("");

const numbers = Array.from({length:21}, (_,i)=>i);

const shapes = ["●", "■", "▲", "★", "♦", "♥"];

const colors = ["#e74c3c", "#3498db", "#2ecc71", "#f39c12", "#9b59b6", "#e91e63", "#00bcd4", "#ff5722", "#8bc34a", "#673ab7"];

let currentLetter = "";

let welcomeSound = new Audio("welcome.ogg");

let bgMusic = new Audio("music.ogg");

bgMusic.loop = true;

bgMusic.volume = 0.06;

let audioUnlocked = false;

let musicEnabled = true;

let welcomePlayed = false;

let letterChoices = 4;

let maxShapes = 5;


function getLetterFile(letter) {

  return letter === "Ё" ? "letter_yo.ogg" : `letter_${letter}.ogg`;

}


function unlockAudio() {

  if (audioUnlocked) return;

  audioUnlocked = true;

  new Audio().play().catch(()=>{});

}


function playSound(name) {

  if (!audioUnlocked) return null;

  const a = new Audio(name);

  a.play().catch(()=>{});

  return a;

}


function playSuccess() {

  const n = Math.floor(Math.random() * 5) + 1;

  return playSound(`success${n}.ogg`);

}


function showGear(show) {

  document.getElementById("gear").style.display = show ? "flex" : "none";

  if (!show) document.getElementById("settings").style.display = "none";

}


function showBack(show) {

  document.getElementById("back-btn").style.display = show ? "block" : "none";

}


function toggleSettings() {

  const panel = document.getElementById("settings");

  panel.style.display = panel.style.display === "block" ? "none" : "block";

}


function toggleMusic() {

  musicEnabled = !musicEnabled;

  document.getElementById("music-btn").textContent = musicEnabled ? "Вкл" : "Выкл";

  document.getElementById("music-btn").style.background = musicEnabled ? "#5cb85c" : "#d9534f";

  if (musicEnabled && document.getElementById("games").style.display === "block") {

    playBgMusic();

  } else {

    stopMusic();

  }

}


function playBgMusic() {

  if (!musicEnabled || !audioUnlocked) return;

  stopMusic();

  bgMusic.currentTime = 0;

  bgMusic.play().catch(()=>{});

}


function stopMusic() {

  bgMusic.pause();

  bgMusic.currentTime = 0;

  welcomeSound.pause();

  welcomeSound.currentTime = 0;

}


function changeCount(type, delta) {

  if (type === "letter") {

    letterChoices = Math.max(3, Math.min(10, letterChoices + delta));

    document.getElementById("letter-display").textContent = letterChoices;

  } else {

    maxShapes = Math.max(1, Math.min(12, maxShapes + delta));

    document.getElementById("shapes-display-setting").textContent = maxShapes;

  }

}


function startWelcome() {

  unlockAudio();

  document.getElementById("menu").style.display = "none";

  document.getElementById("games").style.display = "block";

  showGear(true);

  showBack(false);


  if (!welcomePlayed) {

    welcomePlayed = true;

    welcomeSound.play().catch(()=>{});

    welcomeSound.addEventListener("ended", () => {

      if (musicEnabled) playBgMusic();

    }, { once: true });

  } else {

    if (musicEnabled) playBgMusic();

  }

}


function startGame(game) {

  stopMusic();

  showGear(false);

  showBack(true);

  document.getElementById("games").style.display = "none";

  document.getElementById(game).style.display = "block";

  if (game === "letters") startLetters();

  if (game === "numbers") startNumbers();

  if (game === "alphabet") buildAlphabet();

  if (game === "count") buildCount();

}


function backToMenu() {

  ["letters","numbers","alphabet","count"].forEach(id => document.getElementById(id).style.display = "none");

  document.getElementById("games").style.display = "block";

  showGear(true);

  showBack(false);

  if (musicEnabled) playBgMusic();

}


function startLetters() {

  currentLetter = letters[Math.floor(Math.random()*letters.length)];

  document.getElementById("repeat-btn").textContent = "Повтори";

  document.getElementById("repeat-btn").onclick = playCurrentLetter;

  buildLetterChoices();

  const question = playSound("find_letter.ogg");

  if (question) {

    question.addEventListener("ended", () => setTimeout(playCurrentLetter, 200), { once: true });

  } else {

    setTimeout(playCurrentLetter, 1800);

  }

}


function playCurrentLetter() {

  playSound(getLetterFile(currentLetter));

}


function buildLetterChoices() {

  const container = document.getElementById("letter-choices");

  container.innerHTML = "";

  const choices = [currentLetter];

  while (choices.length < letterChoices) {

    const l = letters[Math.floor(Math.random()*letters.length)];

    if (!choices.includes(l)) choices.push(l);

  }

  choices.sort(() => Math.random()-0.5);

  choices.forEach(l => {

    const div = document.createElement("div");

    div.className = "letter";

    div.textContent = l;

    div.onclick = () => {

      if (l === currentLetter) {

        playSuccess();

        document.getElementById("repeat-btn").textContent = "Продолжить";

        document.getElementById("repeat-btn").onclick = startLetters;

      } else {

        playSound("wrong.ogg");

      }

    };

    container.appendChild(div);

  });

}


function startNumbers() {

  playSound("how_many.ogg");

  const count = Math.floor(Math.random() * maxShapes) + 1;

  const shape = shapes[Math.floor(Math.random() * shapes.length)];

  const color = colors[Math.floor(Math.random() * colors.length)];

  let shapesHtml = "";

  for (let i = 0; i < count; i++) {

    shapesHtml += `<div class="shape-box" style="color:${color}">${shape}</div>`;

  }

  document.getElementById("number-question").innerHTML =

    `<h2>Сколько фигур?</h2><div id="shapes-container">${shapesHtml}</div>`;

  const container = document.getElementById("number-choices");

  container.innerHTML = "";

  let opts = [count];

  while (opts.length < 4) {

    const x = Math.floor(Math.random() * 12) + 1;

    if (!opts.includes(x)) opts.push(x);

  }

  opts.sort(() => Math.random()-0.5).forEach(num => {

    const div = document.createElement("div");

    div.className = "num";

    div.textContent = num;

    div.onclick = () => {

      if (num === count) {

        const success = playSuccess();

        if (success) {

          success.addEventListener("ended", () => startNumbers(), { once: true });

        } else {

          setTimeout(startNumbers, 800);

        }

      } else playSound("wrong.ogg");

    };

    container.appendChild(div);

  });

}


function buildAlphabet() {

  const container = document.getElementById("alphabet-grid");

  container.innerHTML = "";

  letters.forEach(l => {

    const div = document.createElement("div");

    div.className = "letter";

    div.textContent = l;

    div.onclick = () => playSound(getLetterFile(l));

    container.appendChild(div);

  });

}


function buildCount() {

  const container = document.getElementById("count-grid");

  container.innerHTML = "";

  numbers.forEach(n => {

    const div = document.createElement("div");

    div.className = "num";

    div.textContent = n;

    div.onclick = () => playSound(`number_${n}.ogg`);

    container.appendChild(div);

  });

}


document.getElementById("menu").style.display = "block";

showGear(true);

showBack(false);

</script>
